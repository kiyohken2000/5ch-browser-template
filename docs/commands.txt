// build: develop
eas build --profile development --platform ios
eas build --profile development --platform android

// build: internal
eas build --profile preview --platform ios
eas build --profile preview --platform android

// build: production
eas build --profile production --platform ios
eas build --profile production --platform android

// build: production (ローカルビルド。EAS のビルド枠を使わない。apps/mobile で実行)
// 成果物は build-*.ipa / build-*.aab。提出は eas submit --platform <ios|android> --path <ファイル>
// iOS はスクリプト経由で実行する。ログインキーチェーンの別の配布用証明書を Xcode が選んで署名に失敗するのを避け、
// fastlane のタイムアウトも延ばす (詳細はスクリプト冒頭のコメント)
./scripts/build-ios-release.sh
eas build --profile production --platform android --local

// build: develop (iOS Simulator, ローカルビルド。EAS のビルド枠を使わない)
// 成果物は build-*.tar.gz。展開した Ember.app を xcrun simctl install でインストール
eas build --profile development --platform ios --local
tar -xzf build-*.tar.gz -C ../../_temp

// OTA update
eas update --channel internal --message "Updating the app"
eas update --channel production --message "Updating the app"

// iOS: Setting up ad hoc provisioning
eas device:create
// list all registered devices for your account
eas device:list

// Expo Settings
eas build:configure
eas update:configure

// iOS Simulator: 起動中のシミュレータと UDID を確認
xcrun simctl list devices booted
// シミュレータを起動
xcrun simctl boot <UDID>
// シミュレータ用 .app をインストール (上書き。データは残る)
xcrun simctl install booted ../../_temp/Ember.app
// リポジトリのルートから
xcrun simctl install booted _temp/Ember.app
// apps/mobile から（commands.txt のまま）
cd apps/mobile && xcrun simctl install booted ../../_temp/Ember.app
// アプリの起動 / 終了
xcrun simctl launch booted net.votepurchase.ember
xcrun simctl terminate booted net.votepurchase.ember
// アプリの削除 (データも消える)
xcrun simctl uninstall booted net.votepurchase.ember
